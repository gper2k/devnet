1. 기본 데이터 바인딩 JS

common.makeTree()로 트리 인스턴스를 생성하고 render(selector, data)로 바인딩합니다. 폴더 아이콘 클릭으로 열기/닫기, 텍스트 클릭으로 선택/해제(토글)합니다.

선택된 노드: 없음
// 트리 생성
const tree = common.makeTree({
    onSelect: (mode, key, node) => {
        console.log(mode, key, node.text);
    }
});

// 데이터 바인딩
tree.render('#tree1', [
    { key: 'root', text: '전체 사용자', child: [
        { key: 'hq', text: '본사 (HQ)', child: [
            { key: 'mgmt', text: '경영기획본부', child: [
                { key: 'strategy', text: '경영전략팀' },
                { key: 'finance', text: '재무관리팀' }
            ]},
            { key: 'rd', text: '연구개발본부 (R&D)', child: [
                { key: 'platform', text: '플랫폼개발팀' },
                { key: 'service', text: '서비스운영팀' }
            ]},
            { key: 'audit', text: '감사실' }
        ]}
    ]}
]);

2. 확장 메뉴 (Context Menu) JS

menus 옵션에 메뉴 항목을 정의하면 각 행 우측에 ⋯ 아이콘이 표시됩니다. 호버 시 나타나며, 클릭 시 드롭 메뉴가 오버레이로 열립니다.

선택된 노드: 없음
const tree = common.makeTree({
    menus: [
        { text: '하위 부서 추가', icon: 'fi fiFolderAdd',
          action: (node, el, tree) => { /* 추가 로직 */ }},
        { text: '이름 변경', icon: 'fi fiEdit',
          action: (node, el, tree) => { /* 수정 로직 */ }},
        { divider: true },
        { text: '삭제', icon: 'fi fiDelete', danger: true,
          action: (node, el, tree) => { /* 삭제 로직 */ }}
    ],
    onSelect: (mode, key, node) => {
        console.log(mode, key, node.text);
    }
});

tree.render('#tree2', data);

3. 옵션 설정 (전체 열기 / 세션 저장 / 체크박스) JS

useAllOpen으로 초기 전체 열기, openSaveKey로 세션 스토리지에 열림 상태 자동 저장/복원, useCheck로 체크박스 모드를 활성화합니다.

전체 열기 + 세션 저장
체크박스 모드
체크된 항목: 없음
// 전체 열기 + 세션 저장
const tree = common.makeTree({
    useAllOpen: true,
    openSaveKey: 'myTree_opens'
});
tree.render('#el', data);

// 체크박스 모드
const checkTree = common.makeTree({
    useCheck: true,
    useOpenEvent: false
});
checkTree.render('#el', data);
const checked = checkTree.getChecked(); // ['key1', 'key2']

4. API 메서드 JS

프로그래밍 방식으로 열기/닫기/선택 제어 및 이벤트 수신.

// 트리 인스턴스 메서드
tree.openAll();            // 전체 펼치기
tree.closeAll();           // 전체 접기
tree.setOpen('key', true); // 특정 노드 열기
tree.setOpen('key', false);// 특정 노드 닫기
tree.clearSelect();        // 선택 해제
tree.getSelect();          // 선택된 키 반환
tree.getSelectNode();      // 선택된 노드 데이터 반환
tree.getChecked();         // 체크된 키 배열 반환

// CustomEvent 수신
document.querySelector('#tree')
    .addEventListener('selected', e => console.log(e.detail.target));
document.querySelector('#tree')
    .addEventListener('unselected', e => console.log(e.detail.target));
document.querySelector('#tree')
    .addEventListener('opened', e => console.log(e.detail.target));
document.querySelector('#tree')
    .addEventListener('closed', e => console.log(e.detail.target));
